import (
	"context"
	"encoding/json"
	"errors"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/t-yang-only/OctoNexus/internal/db"
	"github.com/t-yang-only/OctoNexus/internal/model"
	"gorm.io/gorm"
)

// Review evidence only: these assertions describe defects observed at 6b837e4.
// A PASS confirms the defect is reproduced; this is not a regression acceptance suite.
func TestReviewSub2APIObservations(t *testing.T) {
	ctx := context.Background()
	setup := func(t *testing.T) *gorm.DB {
		t.Helper()
		useImportCipherKey(t, "review-only-test-key")
		conn := openImportTestDB(t)
		if err := conn.AutoMigrate(&model.Sub2APIAccount{}); err != nil {
			t.Fatal(err)
		}
		restore := db.SetDBForTest(conn)
		t.Cleanup(restore)
		return conn
	}
	request := func(name string) Sub2APIAccountCreateRequest {
		return Sub2APIAccountCreateRequest{Name: name, Platform: model.Sub2APIPlatformOpenAI, BaseURL: "https://example.invalid", ApiKey: "review-fake-key", Enabled: true, SyncChannel: true}
	}
	t.Run("disabled_account_becomes_enabled", func(t *testing.T) {
		setup(t)
		r := request(t.Name())
		r.Enabled = false
		a, err := Sub2APIAccountCreate(ctx, r)
		if err != nil {
			t.Fatal(err)
		}
		if !a.Enabled {
			t.Fatal("observation changed: false was preserved")
		}
		t.Log("enabled:false was returned as enabled:true")
	})
	t.Run("materialized_channel_has_no_route", func(t *testing.T) {
		conn := setup(t)
		r := request(t.Name())
		r.APIProtocol = "responses"
		a, err := Sub2APIAccountCreate(ctx, r)
		if err != nil || a.ChannelID == 0 {
			t.Fatalf("create: account=%+v err=%v", a, err)
		}
		for _, m := range []any{&model.ChannelModel{}, &model.ChannelGrant{}, &model.GroupItem{}} {
			var n int64
			if err := conn.Model(m).Count(&n).Error; err != nil {
				t.Fatal(err)
			}
			if n != 0 {
				t.Fatalf("observation changed: %T rows=%d", m, n)
			}
		}
		t.Log("channel created, but model/grant/group-item counts are all zero")
	})
	t.Run("duplicate_name_silently_skips_channel", func(t *testing.T) {
		setup(t)
		r := request(t.Name())
		first, err := Sub2APIAccountCreate(ctx, r)
		if err != nil || first.ChannelID == 0 {
			t.Fatalf("first create: %v", err)
		}
		second, err := Sub2APIAccountCreate(ctx, r)
		if err != nil || second.ChannelID != 0 {
			t.Fatalf("observation changed: second=%+v err=%v", second, err)
		}
		t.Log("same name: second account succeeds with channel_id=0")
	})
	t.Run("delete_error_leaves_channel_alive", func(t *testing.T) {
		conn := setup(t)
		a, err := Sub2APIAccountCreate(ctx, request(t.Name()))
		if err != nil || a.ChannelID == 0 {
			t.Fatalf("create: %v", err)
		}
		if err := conn.Callback().Delete().Before("gorm:delete").Register("review:block_channel_delete", func(tx *gorm.DB) {
			if tx.Statement.Table == "channels" {
				tx.AddError(errors.New("review injected channel deletion failure"))
			}
		}); err != nil {
			t.Fatal(err)
		}
		if err := Sub2APIAccountDelete(ctx, a.ID); err != nil {
			t.Fatalf("observation changed: %v", err)
		}
		var accounts, channels int64
		conn.Model(&model.Sub2APIAccount{}).Count(&accounts)
		conn.Model(&model.Channel{}).Count(&channels)
		if accounts != 0 || channels != 1 {
			t.Fatalf("accounts=%d channels=%d", accounts, channels)
		}
		t.Log("delete reports success: account gone, enabled channel survives")
	})
	t.Run("probe_ignores_proxy_headers_and_http_failure", func(t *testing.T) {
		setup(t)
		var path, custom string
		srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			path, custom = r.URL.Path, r.Header.Get("X-Review-Required")
			w.WriteHeader(http.StatusUnauthorized)
		}))
		defer srv.Close()
		r := request(t.Name())
		r.BaseURL = srv.URL + "/v1"
		r.ProxyNodeID = 99999
		r.SyncChannel = false
		r.Headers = map[string]string{"X-Review-Required": "yes"}
		a, err := Sub2APIAccountCreate(ctx, r)
		if err != nil {
			t.Fatal(err)
		}
		status, err := Sub2APIAccountProbe(ctx, a.ID)
		if err != nil || !strings.Contains(status, "401") || path != "/v1/v1/models" || custom != "" {
			t.Fatalf("status=%s err=%v path=%s custom=%s", status, err, path, custom)
		}
		t.Log("invalid bound proxy was bypassed; header omitted; path=/v1/v1/models; HTTP 401 returned without error")
	})
	t.Run("probe_decryption_failure_is_silent", func(t *testing.T) {
		conn := setup(t)
		var auth string
		srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { auth = r.Header.Get("Authorization"); w.WriteHeader(200) }))
		defer srv.Close()
		r := request(t.Name())
		r.BaseURL = srv.URL
		r.SyncChannel = false
		a, err := Sub2APIAccountCreate(ctx, r)
		if err != nil {
			t.Fatal(err)
		}
		if err := conn.Model(a).Update("credentials", "enc:v1:broken").Error; err != nil {
			t.Fatal(err)
		}
		_, err = Sub2APIAccountProbe(ctx, a.ID)
		if err != nil || auth != "" {
			t.Fatalf("observation changed: err=%v auth present=%v", err, auth != "")
		}
		t.Log("corrupt credentials: probe still sends unauthenticated request and succeeds")
	})
	t.Run("custom_auth_header_is_plaintext_in_response", func(t *testing.T) {
		setup(t)
		r := request(t.Name())
		r.SyncChannel = false
		r.Headers = map[string]string{"Authorization": "Bearer review-fake-header-secret"}
		a, err := Sub2APIAccountCreate(ctx, r)
		if err != nil {
			t.Fatal(err)
		}
		raw, err := json.Marshal(a)
		if err != nil {
			t.Fatal(err)
		}
		if !strings.Contains(a.Extra, "review-fake-header-secret") || !strings.Contains(string(raw), "review-fake-header-secret") {
			t.Fatal("observation changed")
		}
		t.Log("custom authorization header is plaintext in Extra and JSON response")
	})
	t.Run("database_export_omits_new_accounts", func(t *testing.T) {
		setup(t)
		r := request("review-backup-unique-name")
		r.SyncChannel = false
		if _, err := Sub2APIAccountCreate(ctx, r); err != nil {
			t.Fatal(err)
		}
		dump, err := DBExportAll(ctx)
		if err != nil {
			t.Fatal(err)
		}
		raw, err := json.Marshal(dump)
		if err != nil {
			t.Fatal(err)
		}
		if strings.Contains(string(raw), r.Name) || strings.Contains(string(raw), "sub2api_accounts") {
			t.Fatal("observation changed")
		}
		t.Log("DBExportAll succeeds but contains neither account nor sub2api_accounts section")
	})
}
